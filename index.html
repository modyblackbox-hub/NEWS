Check media content or enclosures if "media_content" in entry and entry.media_content: for media in entry.media_content: if media.get("type", "").startswith("image/") or "url" in media: return media["url"] if "enclosures" in entry and entry.enclosures: for enc in entry.enclosures: if enc.get("type", "").startswith("image/"): return enc.get("href") or enc.get("url") # Fallback: Parse HTML summary for tags summary_html = entry.get("summary", entry.get("description", "")) if summary_html: soup = BeautifulSoup(summary_html, "html.parser") img = soup.find("img") if img and img.get("src"): return img["src"] return None def download_image(url): """Download external image and return relative local file path.""" if not url: return None try: # Generate unique local filename based on URL hash url_hash = hashlib.md5(url.encode('utf-8')).hexdigest()[:10] ext = url.split('.')[-1].split('?')[0].lower() if ext not in ['jpg', 'jpeg', 'png', 'webp', 'gif']: ext = 'jpg' filename = f"img_{url_hash}.{ext}" filepath = os.path.join(IMAGE_DIR, filename) relative_path = f"data/images/{filename}" # Skip download if image already exists locally if os.path.exists(filepath): return relative_path # Download image with standard browser User-Agent req = urllib.request.Request(url, headers={'User-Agent': 'Mozilla/5.0 (Windows NT 10.0; Win64; x64)'}) with urllib.request.urlopen(req, timeout=10) as response, open(filepath, 'wb') as out_file: out_file.write(response.read()) return relative_path except Exception as e: print(f"Image download failed for {url}: {e}") return None def calculate_intensity(title, summary): combined = f"{title} {summary}".lower() for level, keywords in INTENSITY_KEYWORDS.items(): if any(kw in combined for kw in keywords): return level return "Standard" def aggregate_news(): all_articles = [] now = datetime.utcnow().strftime("%Y-%m-%d %H:%M:%S UTC") for category, source_list in FEEDS.items(): for source in source_list: try: feed = feedparser.parse(source["url"]) for entry in feed.entries[:5]: title = entry.get("title", "Untitled Article") link = entry.get("link", "#") published = entry.get("published", entry.get("updated", now)) raw_summary = entry.get("summary", entry.get("description", "")) # Clean HTML tags from summary text summary_text = BeautifulSoup(raw_summary, "html.parser").get_text() # Process & cache image locally remote_image = extract_image_url(entry) local_image = download_image(remote_image) if remote_image else None intensity = calculate_intensity(title, summary_text) all_articles.append({ "title": title, "link": link, "source": source["name"], "category": category, "published": published, "summary": summary_text[:250] + "..." if len(summary_text) > 250 else summary_text, "image_url": local_image, "intensity": intensity }) except Exception as e: print(f"Error fetching {source['name']}: {e}") output_data = { "last_updated": now, "total_articles": len(all_articles), "articles": all_articles } with open(os.path.join(DATA_DIR, "news.json"), "w", encoding="utf-8") as f: json.dump(output_data, f, indent=2, ensure_ascii=False) print(f"Successfully updated data/news.json with cached images at {now}") if __name__ == "__main__": aggregate_news() ``` --- ### 2\.
